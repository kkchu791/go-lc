### 21.2-6 No tail approach

<!-- type Node
val int
next *Node
setP *Set

type Set
head *Node
size int -->

UNION(x, y Nodes)

// find there set
sX = FIND-SET(x)
sY = FIND-SET(y)

if sX == sY (these two are already in the same set)
    return 

if sX.size >= sY.size

    // you want to save setX because we'll be using that later. We want to walk the smaller set which is sY
    oldSetX := sX.head
    // reassign sx head to setY because thats gonna be our new start
    sX.head = sY.head

    // now we begin looping through setY and reassign the pointerS to sX

    currEl := sY.head

    for currEl.next != nil {
        currEl.setP = sX
        sX.size++
        currEl = currEl.next
    }

    currEl.setP = sX
    sX.size++
    currEl.next = oldSetX

else {
    // switch x and y in above implementation
    //you'll be walking through setX

    oldSetY := sY.head
    sY.head = sX.head

    currEl := sX.head

    for currEl.next != nil {
        currEl.setP = sY
        currEl = currEl.next
        sY.size++
    }
    currEl.setP = sY
    sY.size++
    currEl.next = oldSetY
}
