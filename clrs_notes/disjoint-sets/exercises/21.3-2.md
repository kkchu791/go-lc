Node x

make-set(x *Node)
x.parent = x
x.rank = 0

find-set(x *Node)
currEl = x
stack = []
while currEl does not equal currEl.parent
    stack := append(stack, curlEl)
    currEl = currEl.parent

root = currEl

while stack is not empty
  node := stack.pop
  node.parent = root

return root

currEl = x
for (currEl != currEl.parent)
    currEl = currEl.parent

// we go up again
// we cache old parent, change the parent, then use old parent to go up

root = currEl

currEl = x
for (currEl != root)
    next = currEl.parent
    currEl.parent = root
    currEl = next
return root
