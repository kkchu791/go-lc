nonrecursiv version of find-set w/ path compression

Node
val int
parent *Node
rank int

Node x
val 0
parent nil
rank nil

make-set(x *Node)
x.parent = x
x.rank = 0

find-set(x *Node)
currEl = x
stack = []
while currEl does not equal currEl.parent
    stack := append(stack, curlEl)
    currEl = currEl.parent

root = currEl

while stack is not empty
  node := stack.pop
  node.parent = root
